country_event = {
	id = fra.1
	title = "An offer from Vichy"
	desc = lar_end_the_occupation.4.desc
	picture = GFX_report_event_degaulle_inspect_troops
	timeout_days = 14
	
	is_triggered_only = yes
	
	option = { #
		name = fra.1.a
		ai_chance = { base = 70 }
		add_to_faction = FROM
		FROM = {
			annex_country = { target = FRA transfer_troops = yes }	
		}
		every_unit_leader = {
			set_nationality = FROM
		}
		custom_effect_tooltip = GAME_OVER_TT
		hidden_effect = {
			FROM = {
				hidden_effect = {
					remove_country_leader_role = {
					    character = FRA_charles_de_gaulle
					    ideology = gaullism
					}		
					add_country_leader_role = {
						character = FRA_charles_de_gaulle 
						promote_leader = yes
						country_leader = {
							ideology=conservatism
							desc = "POLITICS_CHARLES_DE_GAULLE_DESC"
							expire="1971.1.1"
							traits = { nationalist_symbol }
						}
					}								
				}
			}
		}
	}

	option = { #
		name = fra.1.b
		ai_chance = { base = 30 modifier = { FROM = { is_ai = no } factor = 0 } }
		FROM = { country_event = lar_end_the_occupation.6 }
	}
}
