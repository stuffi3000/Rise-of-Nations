# The amicable separation, floor crisis
country_event = {
	id = auh_ausgleich.20
	title = auh_ausgleich.20.t
	desc = auh_ausgleich.20.d
	picture = GFX_report_event_AUH_minorities

	is_triggered_only = yes

	# A - Grant it
	option = {
		name = auh_ausgleich.20.a
		ai_chance = {
			factor = 30
		}
		set_country_flag = AUH_dual_monarchy_dissolved
		# release_puppet: not if HUN already exists
		if = {
			limit = { NOT = { country_exists = HUN } }
			release_puppet = HUN
		}
		set_variable = { HUN_support = 70 }
		if = {
			limit = { NOT = { has_idea = AUH_the_personal_union } }
			AUH_wear_the_personal_union = yes
		}
		set_variable = { AUH_ausgleich_tier = 1 }
		set_variable = { AUH_gemeinsamkeit = 0 }
		add_stability = 0.05
		AUH_clamp_nationality_support = yes
	}

	# B - Refuse it
	option = {
		name = auh_ausgleich.20.b
		ai_chance = {
			factor = 30
		}
		set_country_flag = AUH_dual_monarchy_dissolved
		add_war_support = 0.10
		# activate_mission: not if already running
		if = {
			limit = { NOT = { has_active_mission = HUN_uprising } }
			activate_mission = HUN_uprising
		}
		subtract_from_variable = { HUN_support = 15 }
		add_to_variable = { AUS_support = 5 }
		AUH_clamp_nationality_support = yes
	}

	# C - Pay for it
	option = {
		name = auh_ausgleich.20.c
		ai_chance = {
			factor = 40
		}
		trigger = {
			has_political_power > 100
		}
		add_political_power = -100
		set_variable = { AUH_quote = 50 }
		add_to_variable = { AUH_gemeinsamkeit = 15 }
		add_to_variable = { HUN_support = 10 }
		# must reset: else the floor crisis never reopens
		set_variable = { AUH_missed_renewals = 0 }
		if = {
			limit = { NOT = { has_idea = AUH_bought_union } }
			add_ideas = AUH_bought_union
		}
		AUH_clamp_nationality_support = yes
		AUH_refresh_the_ausgleich_tier = yes
	}
}
