# the alaska award, october 1903
country_event = {
	id = can_news.101
	title = can_news.h.t
	desc = can_news.h.d

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = CAN
		date > 1903.10.20
		NOT = { date > 1904.4.1 }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		set_variable = { CAN_news_id = 901 }
		set_variable = { CAN_news_scope = 2 }
		CAN_headline = yes
	}
}
