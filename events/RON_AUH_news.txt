# the annexation, october 1908
country_event = {
	id = auh_news.101
	title = auh_news.h.t
	desc = auh_news.h.d

	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = AUH
		has_global_flag = annexation_of_bosnia
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		set_variable = { AUH_news_id = 901 }
		set_variable = { AUH_news_scope = 3 }
		AUH_headline = yes
	}
}
