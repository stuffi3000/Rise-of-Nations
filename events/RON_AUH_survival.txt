	id = auh_survival.1
	title = auh_survival.1.t
	desc = auh_survival.1.d
	picture = GFX_report_event_generic_funeral

	is_triggered_only = yes

	option = {
		name = auh_survival.1.a
		ai_chance = { factor = 70 }
		add_stability = 0.03
		custom_effect_tooltip = auh_survival_the_emperor_recovers_tt
		hidden_effect = {
			set_country_flag = AUH_the_emperor_recovered
		}
	}

	option = {
		name = auh_survival.1.b
		ai_chance = { factor = 30 }
		trigger = {
			NOT = { has_country_flag = AUH_the_emperor_recovered }
		}
		add_political_power = 50
		add_ideas = AUH_the_regency_council
		set_country_flag = AUH_the_regency_is_open
		country_event = { id = auh_survival.2 days = 60 }
	}
}
