########################################################################################
# can_dark.1 - THE BRYCE REPORT, 1907
########################################################################################

country_event = {
	id = can_dark.1
	title = can_dark.1.t
	desc = can_dark.1.d
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes
	fire_only_once = yes

	# (a) historical: report withdrawn, post cut in 1919
	option = {
		name = can_dark.1.a

		set_variable = { CAN_ledger_arg_evidence = 2 }
		CAN_add_ledger = yes
		custom_effect_tooltip = CAN_ledger_evidence_two_tt

		add_ideas = CAN_a_national_crime

		set_country_flag = CAN_bryce_report_buried

		ai_chance = {
			factor = 80
		}
	}

	# (b) fund sanatoria and day schools; reformed, not ended
	option = {
		name = can_dark.1.b

		add_political_power = -150
		add_stability = -0.05

		if = {
			limit = { has_idea = CAN_residential_schools }
			swap_ideas = {
				remove_idea = CAN_residential_schools
				add_idea = CAN_residential_schools_reformed
			}
			else = {
				if = {
					limit = { NOT = { has_idea = CAN_residential_schools_reformed } }
					add_ideas = CAN_residential_schools_reformed
				}
			}
		}

		# opens CAN_treaty_renewal
		set_country_flag = CAN_bryce_report_funded
		custom_effect_tooltip = CAN_treaty_renewal_opened_tt

		ai_chance = {
			factor = 10
		}
	}

	# (c) publish it and let parliament read
	option = {
		name = can_dark.1.c

		add_political_power = -75

		add_timed_idea = {
			idea = CAN_the_department_under_scrutiny
			days = 1825
		}

		set_country_flag = CAN_bryce_report_published

		ai_chance = {
			factor = 10
		}
	}
}

########################################################################################
# can_dark.10 - THE LEDGER PRESENTED, 1 january 1946
#
# under fourteen debt points a commission takes back two
########################################################################################

country_event = {
	id = can_dark.10
	title = can_dark.10.t
	desc = can_dark.10.d
	picture = GFX_report_event_generic_parliament

	is_triggered_only = yes
	fire_only_once = yes

	# royal commission: two points back, no more
	option = {
		name = can_dark.10.a

		trigger = {
			check_variable = { var = CAN_civil_liberties_debt value = 14 compare = less_than }
		}

		add_political_power = -100

		set_variable = { CAN_ledger_arg_debt = -2 }
		CAN_add_ledger = yes
		custom_effect_tooltip = CAN_ledger_debt_remitted_two_tt

		ai_chance = {
			factor = 60
		}
	}

	# do nothing; past thirteen points the only option left
	option = {
		name = can_dark.10.b

		if = {
			limit = {
				check_variable = { var = CAN_civil_liberties_debt value = 14 compare = greater_than_or_equals }
				NOT = { has_idea = CAN_a_shameful_ledger }
			}
			add_ideas = CAN_a_shameful_ledger
			set_country_flag = CAN_shameful_ledger
		}

		ai_chance = {
			factor = 40
		}
	}
}


########################################################################################
# can_dark.11 - THE REPATRIATION ORDERS RESCINDED, 1947
#
# if the property was auctioned, only compensation is left
########################################################################################

country_event = {
	id = can_dark.11
	title = can_dark.11.t
	desc = can_dark.11.d
	picture = GFX_report_event_merchant_ship_01

	is_triggered_only = yes
	fire_only_once = yes

	# rescind the orders
	option = {
		name = can_dark.11.a

		add_stability = 0.03

		set_variable = { CAN_ledger_arg_debt = -1 }
		CAN_add_ledger = yes
		custom_effect_tooltip = CAN_ledger_debt_remitted_one_tt

		473 = {
			if = {
				limit = { has_dynamic_modifier = { modifier = CAN_the_coast_under_order } }
				remove_dynamic_modifier = { modifier = CAN_the_coast_under_order }
			}
		}

		ai_chance = {
			factor = 70
		}
	}

	# return their property, only if it still exists
	option = {
		name = can_dark.11.b

		trigger = {
			NOT = { has_country_flag = CAN_done_the_auction_of_their_property }
		}

		add_political_power = -100
		add_stability = 0.02

		set_variable = { CAN_ledger_arg_debt = -1 }
		CAN_add_ledger = yes
		custom_effect_tooltip = CAN_ledger_debt_remitted_one_tt

		473 = {
			if = {
				limit = { has_dynamic_modifier = { modifier = CAN_the_coast_under_order } }
				remove_dynamic_modifier = { modifier = CAN_the_coast_under_order }
			}
		}

		ai_chance = {
			factor = 30
		}
	}

	# compensate, at triple, there is nothing left to return
	option = {
		name = can_dark.11.c

		trigger = {
			has_country_flag = CAN_done_the_auction_of_their_property
		}

		add_political_power = -300

		set_variable = { CAN_ledger_arg_debt = -1 }
		CAN_add_ledger = yes
		custom_effect_tooltip = CAN_ledger_debt_remitted_one_tt

		custom_effect_tooltip = CAN_the_property_has_been_sold_tt

		ai_chance = {
			factor = 20
		}
	}
}
