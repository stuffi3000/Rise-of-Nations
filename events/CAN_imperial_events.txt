########################################################################################
##### 4 August 1914 - the King is at war
########################################################################################

country_event = {
	id = can_imperial.3
	title = can_imperial.3.t
	desc = can_imperial.3.d
	picture = GFX_report_event_king_speech

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		original_tag = CAN
		ENG = { has_war = yes }
		NOT = { has_war_with = ENG }
	}

	immediate = {
		hidden_effect = {
			set_country_flag = CAN_imperial_king_is_at_war_seen
		}
	}

	# only one answer left
	option = {
		name = can_imperial.3.a
		ai_chance = { factor = 100 }

		add_war_support = 0.10

		set_variable = { CAN_bt_arg_value = 5 }
		set_variable = { CAN_bt_arg_cause = 6 }
		CAN_add_blood_tax = yes

		# auto-join the british faction
		if = {
			limit = {
				ENG = { is_faction_leader = yes }
				NOT = { is_in_faction_with = ENG }
			}
			ENG = { add_to_faction = PREV }
		}
		else_if = {
			limit = {
				ENG = { NOT = { is_in_faction = yes } }
				NOT = { is_in_faction = yes }
			}
			ENG = {
				create_faction = CAN_faction_the_empire
				add_to_faction = PREV
			}
		}

		set_country_flag = CAN_imperial_the_king_is_at_war

		CAN_imperial_english_canada_rallies = yes
	}
}
