########################################################################################
# CANADA - MILITARY EVENT TRIGGERS
# CAN_war_events.txt is all is_triggered_only, this is the caller
# on_weekly_CAN also runs CAN_reinforcement_deficit, 0-100, cap 60
########################################################################################

on_actions = {

	# ##############################################################################
	# 1. EVENT WINDOWS
	# each block carries its own dates, else a 1936 start sees 1915 events
	# events are fire_only_once, the flag guards only skip the test
	# ##############################################################################

	on_daily_CAN = {
		effect = {

			# ------------------------------------------------------------------
			# can_war.1 - Paardeberg, 18-27 February 1900
			# ------------------------------------------------------------------
			if = {
				limit = {
					date > 1900.2.18
					date < 1900.6.1
					NOT = { has_country_flag = CAN_evt_paardeberg }
				}
				country_event = { id = can_war.1 }
			}

			# ------------------------------------------------------------------
			# can_war.2 - the Ross jams, the first long attrition of the CEF
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1915.4.1
					date < 1919.1.1
					has_country_flag = CAN_done_the_canadian_expeditionary_force
					NOT = { has_country_flag = CAN_ross_withdrawn }
					NOT = { has_country_flag = CAN_ross_kept }
				}
				country_event = { id = can_war.2 }
			}

			# ------------------------------------------------------------------
			# can_war.3 - gas at Ypres, 22 April 1915
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1915.4.22
					date < 1915.9.1
					NOT = { has_country_flag = CAN_blooded_at_ypres }
					NOT = { has_country_flag = CAN_ypres_flank_given }
				}
				country_event = { id = can_war.3 }
			}

			# ------------------------------------------------------------------
			# can_war.4 - the Canadian Corps is formed, 13 September 1915
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1915.9.13
					date < 1916.9.1
					NOT = { has_country_flag = CAN_corps_imperial_command }
					NOT = { has_country_flag = CAN_corps_canadian_staff }
				}
				country_event = { id = can_war.4 }
			}

			# ------------------------------------------------------------------
			# can_war.5 - Currie takes the Corps, 9 June 1917
			# national path only, after Vimy
			# needs the canadian staff claimed in 1915, never on the imperial path
			# ------------------------------------------------------------------
			if = {
				limit = {
					date > 1917.6.9
					date < 1918.6.1
					has_country_flag = CAN_done_the_militia_act_of_1904
					has_country_flag = CAN_done_the_vimy_method
					has_country_flag = CAN_corps_canadian_staff
					NOT = { has_country_flag = CAN_currie_commands_the_corps }
				}
				country_event = { id = can_war.5 }
			}

			# ------------------------------------------------------------------
			# can_war.6 - the khaki election, 17 December 1917
			# the Military Service Act is the only way in
			# no act, no union government and no conscription election
			# ------------------------------------------------------------------
			if = {
				limit = {
					date > 1917.12.17
					date < 1918.6.1
					has_country_flag = CAN_done_the_military_service_act
					NOT = { has_country_flag = CAN_union_government }
					NOT = { has_country_flag = CAN_no_union_government }
				}
				country_event = { id = can_war.6 }
			}

			# ------------------------------------------------------------------
			# can_war.7 - Easter in Quebec, 28 March - 1 April 1918
			# ------------------------------------------------------------------
			if = {
				limit = {
					date > 1918.3.28
					date < 1918.12.1
					has_country_flag = CAN_done_the_military_service_act
					NOT = { has_country_flag = CAN_easter_repressed }
					NOT = { has_country_flag = CAN_easter_negotiated }
				}
				country_event = { id = can_war.7 }
			}

			# ------------------------------------------------------------------
			# can_war.8 - the Hundred Days, 8 August - 11 November 1918
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1918.8.8
					date < 1918.11.11
					has_country_flag = CAN_done_the_canadian_expeditionary_force
					NOT = { has_country_flag = CAN_the_hundred_days_fought }
					NOT = { has_country_flag = CAN_the_boys_came_home }
				}
				country_event = { id = can_war.8 }
			}

			# ------------------------------------------------------------------
			# can_war.9 - C Force sails, 27 October 1941
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1941.10.27
					date < 1942.4.1
					OR = {
						is_in_faction_with = ENG
						has_war_with = GER
					}
					NOT = { has_country_flag = CAN_c_force_sailed }
					NOT = { has_country_flag = CAN_c_force_refused }
				}
				country_event = { id = can_war.9 }
			}

			# ------------------------------------------------------------------
			# can_war.11 - Jubilee, 19 August 1942
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war_with = GER
					date > 1942.8.19
					date < 1943.6.1
					NOT = { has_country_flag = CAN_jubilee_accepted }
					NOT = { has_country_flag = CAN_jubilee_refused }
				}
				country_event = { id = can_war.11 }
			}

			# ------------------------------------------------------------------
			# can_war.12 - the Atlantic crossing, winter 1942-1943
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war_with = GER
					date > 1942.11.1
					date < 1944.1.1
					NOT = { has_country_flag = CAN_atlantic_escorts_first }
					NOT = { has_country_flag = CAN_atlantic_army_first }
				}
				country_event = { id = can_war.12 }
			}

			# ------------------------------------------------------------------
			# can_war.13 - the Scheldt, 2 October - 8 November 1944
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war_with = GER
					date > 1944.10.2
					date < 1945.3.1
					NOT = { has_country_flag = CAN_the_scheldt_cleared }
					NOT = { has_country_flag = CAN_the_scheldt_postponed }
				}
				country_event = { id = can_war.13 }
			}

			# ------------------------------------------------------------------
			# can_war.14 - the reinforcement crisis
			# the counter decides if, the date decides when
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					date > 1944.11.1
					check_variable = {
						var = CAN_reinforcement_deficit
						value = 60
						compare = greater_than_or_equals
					}
					NOT = { has_country_flag = CAN_reinforcement_crisis_open }
				}
				country_event = { id = can_war.14 }
			}

			# ------------------------------------------------------------------
			# can_war.15 - the sixteen thousand, 22-23 November 1944
			# fired by the chosen outcome, never by a date
			# CAN_no_conscription sends nobody, so there is no third call
			# ------------------------------------------------------------------
			if = {
				limit = {
					OR = {
						has_completed_focus = CAN_ralston_stays
						has_completed_focus = CAN_the_king_manoeuvre
					}
					NOT = { has_country_flag = CAN_home_defence_men_overseas }
				}
				country_event = { id = can_war.15 }
			}
		}
	}

	# ##############################################################################
	# 2. WEEKLY TICK
	# ##############################################################################

	on_weekly_CAN = {
		effect = {

			# ------------------------------------------------------------------
			# 2.1 the reinforcement deficit
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_war = yes
					NOT = { has_country_flag = CAN_reinforcement_crisis_open }
					OR = {
						has_completed_focus = CAN_juno_beach
						AND = {
							date > 1943.12.20
							has_completed_focus = CAN_little_stalingrad
						}
						AND = {
							date > 1944.6.6
							has_war_with = GER
						}
					}
				}

				set_variable = { CAN_rd_tick = 2 }

				if = {
					limit = { has_completed_focus = CAN_juno_beach }
					add_to_variable = { CAN_rd_tick = 1 }
				}
				if = {
					limit = { has_completed_focus = CAN_little_stalingrad }
					multiply_variable = { CAN_rd_tick = 1.5 }
				}

				add_to_variable = { CAN_reinforcement_deficit = CAN_rd_tick }
				clamp_variable = { var = CAN_reinforcement_deficit min = 0 max = 100 }
			}

			# ------------------------------------------------------------------
			# 2.2 cause 20 - home defence men overseas, +1 every 30 days
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_country_flag = CAN_home_defence_men_overseas
					has_war = yes
				}
				add_to_variable = { CAN_bt_zombie_days = 7 }

				if = {
					limit = {
						check_variable = {
							var = CAN_bt_zombie_days
							value = 30
							compare = greater_than_or_equals
						}
					}
					subtract_from_variable = { CAN_bt_zombie_days = 30 }
					set_variable = { CAN_bt_arg_value = 1 }
					set_variable = { CAN_bt_arg_cause = 20 }
					CAN_add_blood_tax = yes
				}
			}

			# ------------------------------------------------------------------
			# 2.3 Terrace smothered - desertion is paid every 30 days
			# repressing costs two stability once, smothering costs nothing
			# ------------------------------------------------------------------
			if = {
				limit = {
					has_country_flag = CAN_terrace_smothered
					has_war = yes
				}
				add_to_variable = { CAN_terrace_days = 7 }

				if = {
					limit = {
						check_variable = {
							var = CAN_terrace_days
							value = 30
							compare = greater_than_or_equals
						}
					}
					subtract_from_variable = { CAN_terrace_days = 30 }
					add_manpower = -1000
					add_war_support = -0.01
				}
			}
		}
	}
}
