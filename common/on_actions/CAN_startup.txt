# ##############################################################################

# ##############################################################################

# every era strictly before the start date is seeded
# bounds copied from RON_handle_focus_trees_start
#     DNC  < 1910.1.1
#     WW1  1910.1.1 -> 1918.12.31
#     INT  1919.1.1 -> 1932.12.31
#     WW2  1933.1.1 -> 1948.12.31
#     CW   1949.1.1 -> 1979.12.31
# on a mutually_exclusive pair only the historical side is set
# on_actions from several files stack
# ##############################################################################

	on_startup = {
		effect = {

			CAN = {

				# ######################################################
				# DNC ERA (1900-1909) - 14 flags
				# ######################################################
				if = {
					limit = {

					set_country_flag = CAN_done_the_autonomy_bills
					# industrial disputes investigation act, 1907
					set_country_flag = CAN_done_the_industrial_disputes_act

					set_country_flag = CAN_done_the_militia_act_of_1904
				}

				# ######################################################
				# WW1 ERA (1910-1918) - 10 flags plus one helper
				# ######################################################
				if = {
					limit = {
						has_start_date > 1918.12.31
					}

					# branch 09 - navy
					# naval service act, 4 may 1910
					set_country_flag = CAN_done_the_naval_service_act

					# branch 01 - imperial autonomy
					# no permanent imperial secretariat

					# branch 03 - nationalist press

					set_country_flag = CAN_done_a_paper_of_our_own

					# canadian expeditionary force, from aug 1914
					set_country_flag = CAN_done_the_canadian_expeditionary_force
					# vimy, 9 apr 1917
					set_country_flag = CAN_done_the_vimy_method
					# military service act, 29 aug 1917
					set_country_flag = CAN_done_the_military_service_act

					# branch 06 - war industry
					# imperial munitions board, nov 1915
					set_country_flag = CAN_done_the_imperial_munitions_board

					# branch 04 - agrarianism
					# new national policy, 1918
					set_country_flag = CAN_done_the_new_national_policy

					# gates CAN_resolution_the_ninth
					set_country_flag = can_birth_of_a_nation
				}

				# ######################################################
				# INT ERA (1919-1932) - 40 flags
				# ######################################################
				if = {
					limit = {

					# branch 01 - international status
					# separate signature at versailles, 1919

					set_country_flag = CAN_done_parliament_will_decide
					# halibut treaty, 2 mar 1923
					set_country_flag = CAN_done_the_halibut_precedent
					# dandurand at geneva, 1924
					set_country_flag = CAN_done_a_fireproof_house
					# not bound by lausanne, 1923
					set_country_flag = CAN_done_not_bound_at_lausanne
					# balfour report, 1926
					set_country_flag = CAN_done_equal_in_status
					# governor general acts on canadian advice
					set_country_flag = CAN_done_the_advice_of_canadian_ministers
					# legations, washington 1927, paris and tokyo 1928
					set_country_flag = CAN_done_legations_abroad
					# league council seat, 1927-1930
					set_country_flag = CAN_done_a_seat_at_geneva
					# conference on dominion legislation, 1929
					set_country_flag = CAN_done_the_conference_on_dominion_legislation
					# statute of westminster, 11 dec 1931
					set_country_flag = CAN_done_a_statute_for_westminster
					# section 7, the BNA act stays in london
					set_country_flag = CAN_done_section_seven

					# branch 02 - labour and schools
					# criminal code section 98, after winnipeg 1919
					set_country_flag = CAN_done_section_ninety_eight
					# veterans' twenty dollars a month
					set_country_flag = CAN_done_twenty_dollars_a_month
					# regulation 17 eased in ontario, 1927
					set_country_flag = CAN_done_compromise_on_bilingual_schools

					# branch 03 - third parties
					# western labour conference, mar 1919
					set_country_flag = CAN_done_the_western_labour_conference
					# progressives, 65 seats in 1921
					set_country_flag = CAN_done_a_third_party_in_the_house
					# crerar refuses official opposition
					set_country_flag = CAN_done_refuse_the_front_bench
					# ginger group splits off, 1924
					set_country_flag = CAN_done_the_ginger_group

					set_country_flag = CAN_done_the_price_of_our_votes

					set_country_flag = CAN_done_the_goglu_press

					set_country_flag = CAN_done_the_league_for_social_reconstruction
					# aberhart's prophetic bible institute, 1927
					set_country_flag = CAN_done_the_prophetic_bible_institute
					# woodsworth, unemployment or war
					set_country_flag = CAN_done_unemployment_or_war

					# royal canadian air force, 1 apr 1924
					set_country_flag = CAN_done_the_royal_canadian_air_force
					# northern air surveys, 1920s
					set_country_flag = CAN_done_the_northern_air_survey
					# camp borden then trenton, 1931
					set_country_flag = CAN_done_the_school_at_trenton
					# trans-canada airway, relief works of 1932
					set_country_flag = CAN_done_the_trans_canada_airway

					# branch 07 - navy
					# RNCVR 1914 then RCNVR 1923
					set_country_flag = CAN_done_the_volunteer_reserve
					# first ships built at home for the RCN
					set_country_flag = CAN_done_the_first_ships_built_for_us
				}

				# ######################################################
				# WW2 ERA (1933-1948) - 5 flags
				# ######################################################
				if = {
					limit = {
						has_start_date > 1948.12.31
					}

					# citizenship act, in force 1 jan 1947
					set_country_flag = CAN_done_a_citizenship_of_our_own
					# st-laurent's gray lecture, 13 jan 1947
					set_country_flag = CAN_done_the_gray_lecture
					# newfoundland referendums, 1948
					set_country_flag = CAN_done_the_newfoundland_question
					# family allowances, first cheques jul 1945
					set_country_flag = CAN_done_family_allowances
					# bank of canada opens, 11 mar 1935
					set_country_flag = CAN_done_a_central_bank_for_the_dominion
				}

				# ######################################################
				# THE BLOOD TAX - language rift counter
				# sets the start value and clamps it
				# defined in common/scripted_effects/CAN_scripted_effects.txt
				# called last so it can read any CAN_done_ flag
				# never starts at zero on a canadian game, hard floor is 10
				# ######################################################
				CAN_initialize_blood_tax = yes
			}
		}
	}

	# ##############################################################
	# WEEKLY BLOOD TAX TICK
	# on_weekly fires for every country, hence the tag check
	#   - decay of -1 per 180 days, at peace and 365 days after a draw
	#   - resync of the seven derived values
	#   - modifiers on states that turned canadian since last week
	# the flag blocks any tick before CAN_initialize_blood_tax
	# ##############################################################
	on_weekly = {
		effect = {
			if = {
				limit = {
					tag = CAN
					has_country_flag = CAN_blood_tax_initialized
				}
				CAN_blood_tax_weekly = yes
			}
		}
	}
}
