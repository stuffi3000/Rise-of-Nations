on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = AUH }
				AUH = { AUH_watch_initialise = yes }
			}
		}
	}

	# initialise here as well, for saves started before the watch existed
	on_weekly = {
		effect = {
			if = {
				limit = { tag = AUH }
				AUH_watch_initialise = yes
				AUH_watch_weekly = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { tag = AUH }
				AUH_watch_monthly = yes
			}
		}
	}
}
