on_actions = {
	on_weekly = {
		effect = {
			if = {
				limit = {
					country_exists = ROOT
					check_variable = {
						num_core_states < 1
					}
				}
				log = "[GetYear] [GetMonth] | DEBUG | [ROOT.GetName] | Has no core but exists"
			}
			if = {
				limit = {
					has_idea = AUH_CZE_nationalism
				}
				subtract_from_variable = { CZE_support = 1 }
			}
			# a nationality decision runs thirty days and pays a point a week while it does
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_SER_support
				}
				add_to_variable = { SER_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_CRO_support
				}
				add_to_variable = { CRO_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_BOS_support
				}
				add_to_variable = { BOS_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_AUS_support
				}
				add_to_variable = { AUS_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_CZE_support
				}
				add_to_variable = { CZE_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_HUN_support
				}
				add_to_variable = { HUN_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_POL_support
				}
				add_to_variable = { POL_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_ROM_support
				}
				add_to_variable = { ROM_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_SLV_support
				}
				add_to_variable = { SLV_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_decision = decrease_UKR_support
				}
				add_to_variable = { UKR_support = 1 }
			}
			if = {
				limit = {
					tag = AUH
					has_country_flag = AUH_ausgleich_open
				}
				AUH_clamp_nationality_support = yes
			}
			if = {
				limit = {
					date < 1920.1.1
					ROOT = {
						has_country_flag = ww1_famine_risk
					}

				}
				#calculate the new consumption
				set_variable = { debug = num_battalions }
				set_temp_variable = { temp1 = num_battalions }
				multiply_temp_variable = { temp1 = ration_size?7 }
				divide_temp_variable = { temp1 = 1000 }
				set_variable = { weekly_grain_consumption = temp1 }
				clamp_variable = {
					var = weekly_grain_consumption
					min = 0.01
					max = 1000000

				}


				#grain consumption is in kt
				subtract_from_variable = { grain_stockpile = weekly_grain_consumption }

				#days left

				set_temp_variable = { temp2 = grain_stockpile }
				divide_temp_variable = { temp2 = weekly_grain_consumption }
				multiply_temp_variable = { temp2 = 7 }
				set_variable = { grain_stockpile_days_left = temp2 }

				clamp_variable = {
					var = grain_stockpile
					min = 0

				}
				clamp_variable = {
					var = grain_stockpile_days_left
					min = 0
					max = 999999
				}

				clamp_variable = {
					var = ration_size
					min = 3.5

				}
				if = {
					limit = {
						check_variable = { grain_stockpile_days_left < 10 }

					}
					add_ideas = famine_3
					if = {
						limit = { has_idea = famine_2 }
						remove_ideas = famine_2
					}
					if = {
						limit = {
							has_idea = famine_1
						}
						remove_ideas = famine_1
					}
					if = {
						limit = {
							has_idea = famine_0
						}
						remove_ideas = famine_0
					}

				}
				else_if = {
					limit = {
						check_variable = { grain_stockpile_days_left > 9 }
						check_variable = { grain_stockpile_days_left < 30 }
					}
					add_ideas = famine_2
					if = {
						limit = {
							has_idea = famine_3
						}
						remove_ideas = famine_3
					}
					if = {
						limit = {
							has_idea = famine_1
						}
						remove_ideas = famine_1
					}
					if = {
						limit = {
							has_idea = famine_0
						}
						remove_ideas = famine_0
					}
				}
				else_if = {
					limit = {
						check_variable = { grain_stockpile_days_left > 29 }
						check_variable = { grain_stockpile_days_left < 180 }

					}
					add_ideas = famine_1

					if = {
						limit = {
							has_idea = famine_3
						}
						remove_ideas = famine_3
					}
					if = {
						limit = {
							has_idea = famine_2
						}
						remove_ideas = famine_2
					}
					if = {
						limit = {
							has_idea = famine_0
						}
						remove_ideas = famine_0
					}
				}
				else_if = {
					limit = {
						check_variable = { grain_stockpile_days_left > 179 }
						check_variable = { grain_stockpile_days_left < 365 }
					}
					add_ideas = famine_0
					if = {
						limit = {
							has_idea = famine_3
						}
						remove_ideas = famine_3
					}
					if = {
						limit = {
							has_idea = famine_2
						}
						remove_ideas = famine_2
					}
					if = {
						limit = {
							has_idea = famine_1
						}
						remove_ideas = famine_1
					}


				}
				else = {
					if = {
						limit = {
							has_idea = famine_3
						}
						remove_ideas = famine_3
					}
					if = {
						limit = {
							has_idea = famine_2
						}
						remove_ideas = famine_2
					}
					if = {
						limit = {
							has_idea = famine_1
						}
						remove_ideas = famine_1
					}
					if = {
						limit = {
							has_idea = famine_0
						}
						remove_ideas = famine_0
					}


				}
			}
		}
	}

	# Austria-Hungary : the settlement has to exist from the first day of the bookmark,
	# not from the first month, because the renewal clock and the map text both read it.
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = AUH }
				AUH = { AUH_initialise_the_ausgleich = yes }
			}
		}
	}

	# FROM is the war target.
	# Every war Vienna declares is charged to the common account, and Budapest re-prices it
	# at the next term. Under a personal union there is no common foreign policy left to
	# declare it with, and Hungary has to be asked first.
	on_declare_war = {
		effect = {
			if = {
				limit = {
					tag = AUH
					has_country_flag = AUH_ausgleich_open
					NOT = { has_country_flag = AUH_dual_monarchy_dissolved }
				}
				add_to_variable = { AUH_quote = 5 }
				clamp_variable = {
					var = AUH_quote
					min = 25
					max = 70
				}
				if = {
					limit = { check_variable = { AUH_ausgleich_tier = 1 } }
					country_event = { id = auh_ausgleich.11 hours = 6 }
				}
			}
		}
	}

	on_monthly = {
		effect = {
			            RON_update_crisis_gui_values = yes

			# the settlement opens once; the tick then moves the community, the quota
			# and the clock. monthly, not weekly, so a core is never lost without warning.
			if = {
				limit = { tag = AUH }
				AUH_initialise_the_ausgleich = yes
				if = {
					limit = {
						has_country_flag = AUH_ausgleich_open
						NOT = { has_country_flag = AUH_dual_monarchy_dissolved }
					}
					AUH_monthly_ausgleich_tick = yes
				}
				if = {
					limit = {
						has_country_flag = AUH_ausgleich_open
					}
					calculate_uprising = yes
				}
			}

			FRA = {
				if = {
					limit = {
						has_idea = FRA_syndicalist_outlets
					}
					
					add_to_variable = {
						syndicalist_influence = 5
					}
					clamp_variable = {
						var = syndicalist_influence
						min = 0
						max = 100
					}
					set_temp_variable = { temp1 = 100 }
					subtract_from_temp_variable = { temp1 = syndicalist_influence }
					set_variable = { antisyndie_influence = temp1 }
				}
				
			}
			every_country = {
				limit = {
					has_country_flag = has_war_with_@ROOT
				}
				add_to_variable = { days_at_war_with_multiplier_@ROOT = 0.2 }
			}

			if = {
				limit = {
					date > 1910.1.1
					date < 1920.1.1
					NOT = {
						has_country_flag = army_dump
					}
					OR = {
						has_idea = famine_0
						has_idea = famine_1
						has_idea = famine_2
						has_idea = famine_3
					}
				}
				army_desertion = yes
			}

		}
	}
}
