#########################################################################
# MONTENEGRO - THE DOWRY AND THE CLAIM
#########################################################################
# MNT_the_dowry - foreign court money
# MNT_the_claim - who owns MNT by law
# vars survive era change, focuses do not
# writes once per run, never resets

MNT_initialise_ledger = {
	if = {
		limit = { NOT = { has_country_flag = MNT_ledger_open } }
		set_country_flag = MNT_ledger_open

		# starting values by bookmark
		if = {
			limit = { date > 1949.1.1 }
			set_variable = { MNT_the_dowry = 0 }
			set_variable = { MNT_the_claim = 95 }
		}
		else_if = {
			limit = { date > 1941.1.1 }
			set_variable = { MNT_the_dowry = 35 }
			set_variable = { MNT_the_claim = 90 }
		}
		else_if = {
			limit = { date > 1933.1.1 }
			set_variable = { MNT_the_dowry = 6 }
			set_variable = { MNT_the_claim = 86 }
		}
		else_if = {
			limit = { date > 1919.1.1 }
			set_variable = { MNT_the_dowry = 28 }
			set_variable = { MNT_the_claim = 72 }
		}
		else_if = {
			limit = { date > 1910.1.1 }
			set_variable = { MNT_the_dowry = 70 }
			set_variable = { MNT_the_claim = 24 }
		}
		else = {
			set_variable = { MNT_the_dowry = 62 }
			set_variable = { MNT_the_claim = 18 }
		}
	}
}
