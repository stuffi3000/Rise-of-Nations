########################################################################################
# THE STATE OF EXCEPTION
########################################################################################
# disarming alone only defers it
# so each block lifts the register clear

AUH_settlement_disperse_the_risings = {
	AUH_settlement_count_the_risings = yes
	if = {
		limit = { has_active_mission = serbian_uprising }
		remove_mission = serbian_uprising
		if = {
			limit = { check_variable = { SER_support < 12 } }
			set_variable = { SER_support = 12 }
		}
		every_state = {
			limit = {
				is_controlled_by = AUH
				is_core_of = SER
			}
			add_compliance = 10
			add_resistance = -10
		}
	}
	if = {
		limit = { has_active_mission = croatian_uprising }
		remove_mission = croatian_uprising
		if = {
			limit = { check_variable = { CRO_support < 12 } }
			set_variable = { CRO_support = 12 }
		}
		every_state = {
			limit = {
				is_controlled_by = AUH
				is_core_of = CRO
			}
			add_compliance = 10
			add_resistance = -10
		}
	}
	if = {
		limit = { has_active_mission = bosnian_uprising }
		remove_mission = bosnian_uprising

			limit = {
				is_controlled_by = AUH
				is_core_of = CZE

			add_compliance = 10

		remove_mission = HUN_uprising

			limit = {
				has_idea = AUH_hungarian_national_resistance
				check_variable = { HUN_support < 30 }
			}
			set_variable = { HUN_support = 30 }
		}
		if = {
			limit = { check_variable = { HUN_support < 12 } }
			set_variable = { HUN_support = 12 }
		}

		}
		every_state = {
			limit = {
				is_controlled_by = AUH
				is_core_of = UKR
			}
			add_compliance = 10
			add_resistance = -10
		}
	}
	# two points of Community per province
	set_temp_variable = { AUH_settlement_toll = AUH_settlement_risings }
	multiply_variable = { AUH_settlement_toll = 2 }
	subtract_from_variable = { AUH_gemeinsamkeit = AUH_settlement_toll }
	AUH_settlement_bound_the_registers = yes
