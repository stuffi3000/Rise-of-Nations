# AUH nationality watch
# AUH_<TAG>_seen        last value the weekly pass looked at
# AUH_<TAG>_ref         value at the start of the month
# AUH_<TAG>_fall        points lost over the month, positive going down
# AUH_<TAG>_months_left months before 5 at the current rate, 99 = not falling
# AUH_alarm_level       1 under 50, 2 under 35, 3 under 20, 4 under 5, 5 back over 35
# nothing here writes a support value

AUH_watch_initialise = {
	if = {
		limit = { NOT = { has_country_flag = AUH_watch_open } }
		set_country_flag = AUH_watch_open

		set_variable = { AUH_AUS_seen = AUS_support }
		set_variable = { AUH_AUS_ref = AUS_support }
		set_variable = { AUH_AUS_months_left = 99 }

		set_variable = { AUH_CZE_ref = CZE_support }
		set_variable = { AUH_CZE_months_left = 99 }

		set_variable = { AUH_SLO_ref = SLO_support }

# slope over the month and the forecast; SLO and ITA are unclamped and may run outside 0-100
AUH_watch_monthly = {
	set_temp_variable = { AUH_tmp_fall = AUH_AUS_ref }
	subtract_from_temp_variable = { AUH_tmp_fall = AUS_support }
	set_variable = { AUH_AUS_fall = AUH_tmp_fall }
	if = {
		limit = { check_variable = { AUH_AUS_fall > 0.001 } }
		set_temp_variable = { AUH_tmp_left = AUS_support }
		subtract_from_temp_variable = { AUH_tmp_left = 5 }
		divide_temp_variable = { AUH_tmp_left = AUH_AUS_fall }
		clamp_temp_variable = { var = AUH_tmp_left  min = 0  max = 98 }
		set_variable = { AUH_AUS_months_left = AUH_tmp_left }
		round_variable = AUH_AUS_months_left
	}
	else = { set_variable = { AUH_AUS_months_left = 99 } }
	set_variable = { AUH_AUS_ref = AUS_support }

	set_temp_variable = { AUH_tmp_fall = AUH_HUN_ref }
	subtract_from_temp_variable = { AUH_tmp_fall = HUN_support }
	set_variable = { AUH_HUN_fall = AUH_tmp_fall }
	if = {
		limit = { check_variable = { AUH_HUN_fall > 0.001 } }
		set_temp_variable = { AUH_tmp_left = HUN_support }
		subtract_from_temp_variable = { AUH_tmp_left = 5 }
		divide_temp_variable = { AUH_tmp_left = AUH_HUN_fall }
		clamp_temp_variable = { var = AUH_tmp_left  min = 0  max = 98 }
		set_variable = { AUH_HUN_months_left = AUH_tmp_left }
		round_variable = AUH_HUN_months_left
	}
	else = { set_variable = { AUH_HUN_months_left = 99 } }
	set_variable = { AUH_HUN_ref = HUN_support }

	set_temp_variable = { AUH_tmp_fall = AUH_CZE_ref }
	subtract_from_temp_variable = { AUH_tmp_fall = CZE_support }
	set_variable = { AUH_CZE_fall = AUH_tmp_fall }
	if = {
