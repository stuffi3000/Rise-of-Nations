#########################################################################
# ECUADOR - THE FILE
#########################################################################
# ECU_the_title - legal case, 0-100
# ECU_effective_possession - ground held
# ECU_guarantors - powers behind it, 0-4
# vars survive era change, focuses do not
# writes once per run, never resets

ECU_initialise_the_file = {
	if = {
		limit = { NOT = { has_country_flag = ECU_file_open } }
		set_country_flag = ECU_file_open

		if = {
			limit = { date > 1949.1.1 }
			set_variable = { ECU_the_title = 0 }
			set_variable = { ECU_effective_possession = 24 }
			set_variable = { ECU_guarantors = 4 }
		}
		else_if = {
			limit = { date > 1933.1.1 }
			set_variable = { ECU_the_title = 88 }
			set_variable = { ECU_effective_possession = 18 }
			set_variable = { ECU_guarantors = 2 }
		}
		else_if = {
			limit = { date > 1919.1.1 }
			set_variable = { ECU_the_title = 78 }
			set_variable = { ECU_effective_possession = 20 }
			set_variable = { ECU_guarantors = 1 }
		}
		else_if = {
			limit = { date > 1910.1.1 }
			set_variable = { ECU_the_title = 70 }
			set_variable = { ECU_effective_possession = 22 }
			set_variable = { ECU_guarantors = 1 }
		}
		else = {
			set_variable = { ECU_the_title = 65 }
			set_variable = { ECU_effective_possession = 22 }
			set_variable = { ECU_guarantors = 0 }
		}
	}
}
