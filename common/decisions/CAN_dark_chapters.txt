	CAN_raise_the_head_tax = {

		icon = generic_confiscation

		allowed = {
			original_tag = CAN
		}

		visible = {
			date < 1924.1.1
			NOT = { has_country_flag = CAN_done_repeal_the_head_tax }
			check_variable = { var = CAN_head_tax_level value = 2 compare = less_than }
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = CAN_raise_the_head_tax_window_tt
				date < 1923.7.1
				OR = {
					check_variable = { var = CAN_head_tax_level value = 0 compare = equals }
					date > 1903.1.1
				}
			}
		}

		cost = 25

		days_remove = 30

		fire_only_once = no

		complete_effect = {
			custom_effect_tooltip = CAN_raise_the_head_tax_pending_tt
		}

		remove_effect = {

			add_to_variable = { CAN_head_tax_level = 1 }

			# the small gain: five years of quiet on the coast
			if = {
				limit = { has_idea = CAN_anti_asiatic_agitation }
				remove_ideas = CAN_anti_asiatic_agitation
			}
			set_variable = { CAN_head_tax_relief_days = 1825 }

			# the permanent cost
			if = {
				limit = { NOT = { has_idea = CAN_exclusion_by_price } }
				add_ideas = CAN_exclusion_by_price
			}
			CAN_place_exclusion_by_price_coast = yes

			set_variable = { CAN_ledger_arg_debt = 1 }
			CAN_add_ledger = yes
			custom_effect_tooltip = CAN_ledger_debt_one_tt

			set_country_flag = CAN_done_raise_the_head_tax
		}

		ai_will_do = {
			factor = 20
		}
	}
