# ##############################################################################
# CAN_decision_categories.txt - the four canadian decision categories
# ##############################################################################
# every canadian decision goes in one of these four
# an unknown category breaks the whole common/decisions load
# the engine hides a category with no visible decision
# allowed gates the nation, decisions gate content
# the west closes for good if the prairies are lost
# no is_ai = no anywhere, the canadian AI plays these too
# icons are base game sprites from interface/decisions.gfx
# ##############################################################################


# ##############################################################################
# 1. IMPERIAL RELATIONS
# ##############################################################################

	allowed = {
		original_tag = CAN
	}
