	# THE RELIEF CAMPS - october 1932, order-in-council
	# each take buys infrastructure and pays in communist support
	# --------------------------------------------------------------------------
	CAN_the_relief_camps = {

		icon = generic_construction

		allowed = {
			original_tag = CAN
		}

		visible = {
			date > 1932.10.8
			date < 1937.1.1
			NOT = { has_country_flag = CAN_relief_camps_closed }
		}

		available = {
			has_war = no
		}

		cost = 25
		fire_only_once = no
		days_re_enable = 120

		complete_effect = {

			add_to_variable = { CAN_relief_camps_taken = 1 }

			random_owned_controlled_state = {
				limit = {
					is_core_of = ROOT
					free_building_slots = {
						building = infrastructure
						size > 0
						include_locked = yes
					}
				}
				add_building_construction = {
					type = infrastructure
					level = 1
					instant_build = yes
				}
			}

			add_stability = -0.02
			add_popularity = {
				ideology = communism
				popularity = 0.02
			}
			add_to_variable = { CAN_labour_grievance = 3 }

			# third take, on to ottawa, june 1935
			if = {
				limit = {
					check_variable = { var = CAN_relief_camps_taken value = 3 compare = greater_than_or_equals }
					NOT = { has_country_flag = CAN_on_to_ottawa_ready }
				}
				set_country_flag = CAN_on_to_ottawa_ready
				add_stability = -0.05
				add_popularity = {
					ideology = communism
					popularity = 0.05
				}
				add_to_variable = { CAN_labour_grievance = 5 }
				add_to_variable = { CAN_civil_liberties_debt = 1 }
				custom_effect_tooltip = CAN_on_to_ottawa_ready_tt
			}

			custom_effect_tooltip = CAN_the_relief_camps_tt
		}

		ai_will_do = {
			factor = 3
		}
	}
