	############################################################################
	# The Compromise and its expiry - the clock.
	#
	# AUH_the_ausgleich_term is the 540-day countdown itself. It opens eighteen
	# months before the year printed on the settlement and it runs whether the
	# Crown is ready or not. AUH_the_decennial_renewal is what the Crown pays to
	# stop it: AUH_quote x 3 in political power, so the more Hungary invoices,
	# the more the renegotiation costs.
	#
	# The pair follows the vanilla AST_insufficient_funds_mission /
	# AST_call_on_fascists pattern: a mission carries the timer and the failure,
	# a decision carries the variable cost and the success.
	############################################################################

	AUH_the_ausgleich_term = {

		icon = generic_decision
		priority = 100

		allowed = {
			original_tag = AUH
		}

		activation = {
			tag = AUH
			has_country_flag = AUH_ausgleich_open
			NOT = { has_country_flag = AUH_dual_monarchy_dissolved }
			check_variable = { AUH_renewal_window > 0 }
		}

		is_good = no

		visible = {
			tag = AUH
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = auh_renewal_concluded_tt
				has_country_flag = AUH_ausgleich_renewed
			}
		}

		days_mission_timeout = 540
		days_re_enable = 0
		days_remove = 0
		fire_only_once = no

		complete_effect = {
			clr_country_flag = AUH_ausgleich_renewed
		}

		timeout_effect = {
			AUH_the_renewal_lapses = yes
			country_event = { id = auh_ausgleich.12 hours = 6 }
		}

		ai_will_do = {
			factor = 200
		}
	}

	AUH_the_decennial_renewal = {

		icon = generic_decision

		allowed = {
			original_tag = AUH
		}

		visible = {
			tag = AUH
			has_active_mission = AUH_the_ausgleich_term
		}

		cost = var:AUH_renewal_cost

		available = {
			NOT = { has_country_flag = AUH_dual_monarchy_dissolved }
			custom_trigger_tooltip = {
				check_variable = { AUH_renewal_window > 0 }
				tooltip = auh_renewal_window_open_tt
			}
		}

		fire_only_once = no
		days_re_enable = 0

		complete_effect = {
			set_country_flag = AUH_ausgleich_renewed
			AUH_the_renewal_succeeds = yes
		}

		ai_will_do = {
			factor = 200
		}
	}
