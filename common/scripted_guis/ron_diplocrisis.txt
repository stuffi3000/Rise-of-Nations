scripted_gui = {

    # Top bar button
    crisis_topbar_button = {
        context_type = player_context
        window_name = "crisis_topbar_button_container"
        
        visible = {
            always = yes
        }
        
        effects = {
            open_crisis_window_button_click = {
                if = {
                    limit = { has_country_flag = crisis_window_open }
                    clr_country_flag = crisis_window_open
                }
                else = {
                    set_country_flag = crisis_window_open
                }
            }
        }
        
        triggers = {
            crisis_warning_icon_visible = {
                check_variable = { global.political_active_crisis_list^num > 0 }
            }
            crisis_count_text_visible = {
                check_variable = { global.political_active_crisis_list^num > 0 }
            }
        }
        
        properties = {
            crisis_warning_icon = {
                frame = 2
            }
        }
    }

    # Main crisis window
    crisis_main_window_gui = {
        context_type = player_context
        window_name = "crisis_main_window"
        
        visible = {
            has_country_flag = crisis_window_open
        }
        
        effects = {
            close_button_click = {
                clr_country_flag = crisis_window_open
            }
        }
        
        triggers = {
            no_crisis_text_visible = {
                check_variable = { global.political_active_crisis_list^num < 1 }
            }
        }
        
        dynamic_lists = {
            crisis_entries = {
                array = global.political_active_crisis_list
                entry_container = crisis_entry
                index = crisis_gui_index
            }
        }
        
        triggers = {
            # Main flags always visible for active entries
            originator_flag_visible = {
                always = yes
            }
            receiver_flag_visible = {
                always = yes
            }
            originator_flag_frame_visible = {
                always = yes
            }
            receiver_flag_frame_visible = {
                always = yes
            }
            
            # Stage icons
            stage_1_visible = {
                always = yes
            }
            stage_2_visible = {
                check_variable = { global.crisis_stage^crisis_gui_index > 0 }
            }
            stage_3_visible = {
                check_variable = { global.crisis_stage^crisis_gui_index > 1 }
            }
            
            # Originator backers
            orig_backer_0_visible = {
                check_variable = { global.crisis_originator_backers^crisis_gui_index^num > 0 }
            }
            orig_backer_1_visible = {
                check_variable = { global.crisis_originator_backers^crisis_gui_index^num > 1 }
            }
            orig_backer_2_visible = {
                check_variable = { global.crisis_originator_backers^crisis_gui_index^num > 2 }
            }
            orig_backer_more_visible = {
                check_variable = { global.crisis_originator_backers^crisis_gui_index^num > 3 }
            }
            
            # Receiver backers
            recv_backer_0_visible = {
                check_variable = { global.crisis_receiver_backers^crisis_gui_index^num > 0 }
            }
            recv_backer_1_visible = {
                check_variable = { global.crisis_receiver_backers^crisis_gui_index^num > 1 }
            }
            recv_backer_2_visible = {
                check_variable = { global.crisis_receiver_backers^crisis_gui_index^num > 2 }
            }
            recv_backer_more_visible = {
                check_variable = { global.crisis_receiver_backers^crisis_gui_index^num > 3 }
            }
        }
        
        properties = {
            # Main party flags
            originator_flag = {
                image = "[?global.crisis_originators^crisis_gui_index.GetFlag]"
            }
            receiver_flag = {
                image = "[?global.crisis_receivers^crisis_gui_index.GetFlag]"
            }
            
            # Stage icons
            stage_1 = {
                frame = 1
            }
            stage_2 = {
                frame = 2
            }
            stage_3 = {
                frame = 3
            }
            
            # Progress bar
            progress_bar = {
                frame = global.crisis_progress^crisis_gui_index
            }
            
            # Strength comparison (uses warscore bar - 50 is center)
            strength_compare_bar = {
                frame = global.crisis_strength_ratio^crisis_gui_index
            }
            
            # Originator backer flags
            orig_backer_0 = {
                image = "[?global.crisis_originator_backers^crisis_gui_index.0.GetFlag]"
            }
  
            
            # Receiver backer flags
            recv_backer_0 = {
                image = "[?global.crisis_receiver_backers^crisis_gui_index.0.GetFlag]"
            }

        }
    }
}
